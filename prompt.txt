📱 COMPLETE OFFLINE ML/CV MOBILE APP - COMPREHENSIVE PROMPT FOR CODEX
App Name: "SCANकर"

🎯 PROJECT OVERVIEW
Build a production-ready, 100% offline, zero-cost cross-platform mobile application (iOS & Android) that uses advanced Machine Learning and Computer Vision to scan and digitize handwritten machine reading sheets with table structure preservation. The app must extract data exactly as it appears in the source image - maintaining column structure, row alignment, and table format.
CRITICAL REQUIREMENT: When data is in columns in the image, it MUST be exported in columns. The extracted table structure must be pixel-perfect replica of the source layout.

javascript// When exporting to Excel/PDF/Word:

1. MUST preserve exact column count from source
2. MUST maintain column order (left to right)
3. MUST keep empty cells as empty (not "N/A" or "null")
4. MUST align text (left for names, right for numbers)
5. MUST preserve merged cells if detected
6. MUST maintain consistent column widths
7. MUST add table borders to show structure clearly

// Example Excel export code structure:
const exportToExcel = (extractedData) => {
  const workbook = XLSX.utils.book_new();
  
  // Create 2D array maintaining exact structure
  const worksheetData = [
    extractedData.headers,  // Header row
    ...extractedData.data.map(row => 
      row.cells.map(cell => cell.text)  // Preserve empty strings
    )
  ];
  
  const worksheet = XLSX.utils.aoa_to_sheet(worksheetData);
  
  // Apply formatting
  const range = XLSX.utils.decode_range(worksheet['!ref']);
  
  // Format headers (row 0)
  for (let col = range.s.c; col <= range.e.c; col++) {
    const cellAddress = XLSX.utils.encode_cell({ r: 0, c: col });
    if (worksheet[cellAddress]) {
      worksheet[cellAddress].s = {
        font: { bold: true, color: { rgb: "FFFFFF" } },
        fill: { fgColor: { rgb: "1976D2" } },
        alignment: { horizontal: "center", vertical: "center" }
      };
    }
  }
  
  // Format data cells
  for (let row = range.s.r + 1; row <= range.e.r; row++) {
    for (let col = range.s.c; col <= range.e.c; col++) {
      const cellAddress = XLSX.utils.encode_cell({ r: row, c: col });
      if (worksheet[cellAddress]) {
        worksheet[cellAddress].s = {
          alignment: { 
            horizontal: col === 0 ? "left" : "right",  // Names left, numbers right
            vertical: "center" 
          },
          border: {
            top: { style: "thin", color: { rgb: "000000" } },
            bottom: { style: "thin", color: { rgb: "000000" } },
            left: { style: "thin", color: { rgb: "000000" } },
            right: { style: "thin", color: { rgb: "000000" } }
          }
        };
      }
    }
  }
  
  // Set column widths based on content
  worksheet['!cols'] = extractedData.headers.map((header, idx) => ({
    wch: idx === 0 ? 30 : 12  // Machine name column wider
  }));
  
  XLSX.utils.book_append_sheet(workbook, worksheet, "Machine Readings");
  
  return workbook;
};

javascriptimport OpenCVService from '../cv/OpenCVService';
import TextDetectionModel from '../ml/TextDetectionModel';
import HandwritingOCRModel from '../ml/HandwritingOCRModel';
import TableStructureModel from '../ml/TableStructureModel';
import TableReconstructor from '../table/TableReconstructor';

class OCRPipeline {
  async processImage(imageUri) {
    try {
      // Step 1: Preprocess image with OpenCV
      console.log('🔧 Preprocessing image...');
      const preprocessed = await OpenCVService.preprocessImage(imageUri);
      
      // Step 2: Detect table structure
      console.log('📊 Detecting table structure...');
      const tableStructure = await TableStructureModel.detectTable(preprocessed);
      // Returns: { rows: [...], columns: [...], cells: [...] }
      
      // Step 3: Detect text regions in each cell
      console.log('🔍 Detecting text regions...');
      const textRegions = await TextDetectionModel.detectTextInCells(
        preprocessed,
        tableStructure.cells
      );
      
      // Step 4: Recognize handwriting in each text region
      console.log('✍️ Recognizing handwriting...');
      const recognizedText = await HandwritingOCRModel.recognizeBatch(
        preprocessed,
        textRegions
      );
      
      // Step 5: Reconstruct table with recognized text
      console.log('🏗️ Reconstructing table...');
      const reconstructedTable = await TableReconstructor.buildTable(
        tableStructure,
        recognizedText
      );
      
      // Step 6: Validate and return
      console.log('✅ Validation complete');
      return {
        success: true,
        data: reconstructedTable,
        metadata: {
          rows: tableStructure.rows.length,
          columns: tableStructure.columns.length,
          processingTime: Date.now() - startTime,
          averageConfidence: this.calculateAverageConfidence(recognizedText)
        }
      };
      
    } catch (error) {
      console.error('❌ OCR Pipeline error:', error);
      return {
        success: false,
        error: error.message
      };
    }
  }
  
  calculateAverageConfidence(recognizedText) {
    const total = recognizedText.reduce((sum, item) => sum + item.confidence, 0);
    return total / recognizedText.length;
  }
}

export default new OCRPipeline();

javascriptimport * as XLSX from 'xlsx';
import RNFS from 'react-native-fs';

class ExcelExporter {
  async exportToExcel(extractedData, fileName) {
    try {
      // Create workbook
      const workbook = XLSX.utils.book_new();
      
      // Build 2D array maintaining EXACT structure
      const worksheetData = [
        extractedData.headers,  // Header row
        ...extractedData.data.map(row => 
          row.cells.map(cell => cell.text || '')  // Preserve empty cells
        )
      ];
      
      // Create worksheet from 2D array
      const worksheet = XLSX.utils.aoa_to_sheet(worksheetData);
      
      // Apply formatting
      this.formatWorksheet(worksheet, extractedData);
      
      // Add worksheet to workbook
      XLSX.utils.book_append_sheet(workbook, worksheet, 'Machine Readings');
      
      // Write to file
      const wbout = XLSX.write(workbook, { 
        type: 'base64', 
        bookType: 'xlsx',
        cellStyles: true 
      });
      
      const filePath = `${RNFS.DocumentDirectoryPath}/${fileName}.xlsx`;
      await RNFS.writeFile(filePath, wbout, 'base64');
      
      return {
        success: true,
        filePath
      };
      
    } catch (error) {
      console.error('Excel export error:', error);
      return {
        success: false,
        error: error.message
      };
    }
  }
  
  formatWorksheet(worksheet, extractedData) {
    const range = XLSX.utils.decode_range(worksheet['!ref']);
    
    // Format header row (row 0)
    for (let col = range.s.c; col <= range.e.c; col++) {
      const cellAddress = XLSX.utils.encode_cell({ r: 0, c: col });
      if (worksheet[cellAddress]) {
        worksheet[cellAddress].s = {
          font: { bold: true, sz: 12, color: { rgb: 'FFFFFF' } },
          fill: { fgColor: { rgb: '1976D2' } },
          alignment: { horizontal: 'center', vertical: 'center', wrapText: true },
          border: this.getCellBorder()
        };
      }
    }
    
    // Format data cells
    for (let row = range.s.r + 1; row <= range.e.r; row++) {
      for (let col = range.s.c; col <= range.e.c; col++) {
        const cellAddress = XLSX.utils.encode_cell({ r: row, c: col });
        
        if (!worksheet[cellAddress]) {
          // Create empty cell to maintain structure
          worksheet[cellAddress] = { t: 's', v: '' };
        }
        
        // Get confidence for this cell
        const cellData = extractedData.data[row - 1]?.cells[col];
        const confidence = cellData?.confidence || 0;
        
        worksheet[cellAddress].s = {
          alignment: { 
            horizontal: col === 0 ? 'left' : 'right',  // Machine names left, numbers right
            vertical: 'center',
            wrapText: false
          },
          border: this.getCellBorder(),
          fill: { fgColor: { rgb: this.getConfidenceColor(confidence) } }
        };
      }
    }
    
    // Set column widths
    worksheet['!cols'] = extractedData.headers.map((header, idx) => ({
      wch: idx === 0 ? 35 : 15  // First column wider for machine names
    }));
    
    // Set row heights
    worksheet['!rows'] = Array(range.e.r + 1).fill({ hpt: 20 });
    worksheet['!rows'][0] = { hpt: 25 };  // Header row taller
  }
  
  getCellBorder() {
    return {
      top: { style: 'thin', color: { rgb: '000000' } },
      bottom: { style: 'thin', color: { rgb: '000000' } },
      left: { style: 'thin', color: { rgb: '000000' } },
      right: { style: 'thin', color: { rgb: '000000' } }
    };
  }
  
  getConfidenceColor(confidence) {
    if (confidence >= 0.9) return 'D4EDDA';  // Light green
    if (confidence >= 0.7) return 'FFF3CD';  // Light yellow
    return 'F8D7DA';  // Light red
  }
}

class DatasetLoader:
    def __init__(self, data_dir):
        self.data_dir = data_dir
        self.images = []
        self.annotations = []
    
    def load_dataset(self):
        """Load all training images and annotations"""
        for filename in os.listdir(self.data_dir):
            if filename.endswith('.json'):
                with open(os.path.join(self.data_dir, filename)) as f:
                    data = json.load(f)
                    
                    # Load image
                    img = cv2.imread(data['image'])
                    img = cv2.resize(img, (640, 640))  # Standardize size
                    
                    self.images.append(img)
                    self.annotations.append(data['annotations'])
        
        return np.array(self.images), self.annotations
    
    def preprocess_images(self, images):
        """Preprocess for training"""
        # Normalize to [0, 1]
        images = images.astype('float32') / 255.0
        return images

def build_table_detection_model():
    """Build model to detect table structure"""
    
    inputs = keras.Input(shape=(640, 640, 3))
    
    # Encoder (Feature extraction)
    x = layers.Conv2D(64, 3, activation='relu', padding='same')(inputs)
    x = layers.MaxPooling2D(2)(x)
    x = layers.Conv2D(128, 3, activation='relu', padding='same')(x)
    x = layers.MaxPooling2D(2)(x)
    x = layers.Conv2D(256, 3, activation='relu', padding='same')(x)
    x = layers.MaxPooling2D(2)(x)
    x = layers.Conv2D(512, 3, activation='relu', padding='same')(x)
    
    # Decoder (Table detection)
    x = layers.UpSampling2D(2)(x)
    x = layers.Conv2D(256, 3, activation='relu', padding='same')(x)
    x = layers.UpSampling2D(2)(x)
    x = layers.Conv2D(128, 3, activation='relu', padding='same')(x)
    x = layers.UpSampling2D(2)(x)
    x = layers.Conv2D(64, 3, activation='relu', padding='same')(x)
    
    # Output: Table mask
    outputs = layers.Conv2D(1, 1, activation='sigmoid')(x)
    
    model = keras.Model(inputs=inputs, outputs=outputs)
    return model

def build_handwriting_ocr_model():
    """Build model for handwriting recognition"""
    
    # Input: Image patches of text (128x32)
    inputs = keras.Input(shape=(32, 128, 1))
    
    # CNN layers for feature extraction
    x = layers.Conv2D(32, (3, 3), activation='relu', padding='same')(inputs)
    x = layers.MaxPooling2D((2, 2))(x)
    x = layers.Conv2D(64, (3, 3), activation='relu', padding='same')(x)
    x = layers.MaxPooling2D((2, 2))(x)
    x = layers.Conv2D(128, (3, 3), activation='relu', padding='same')(x)
    x = layers.MaxPooling2D((2, 2))(x)
    
    # Reshape for RNN
    new_shape = ((32, -1))  # (width, features)
    x = layers.Reshape(target_shape=new_shape)(x)
    
    # LSTM layers for sequence modeling
    x = layers.Bidirectional(layers.LSTM(128, return_sequences=True))(x)
    x = layers.Bidirectional(layers.LSTM(128, return_sequences=True))(x)
    
    # Output layer (character probabilities)
    # 37 characters: 0-9, A-Z, space
    outputs = layers.Dense(37, activation='softmax')(x)
    
    model = keras.Model(inputs=inputs, outputs=outputs)
    return model

def convert_to_tflite(model, model_name):
    """Convert Keras model to TFLite"""
    
    # Create converter
    converter = tf.lite.TFLiteConverter.from_keras_model(model)
    
    # Optimize for mobile
    converter.optimizations = [tf.lite.Optimize.DEFAULT]
    converter.target_spec.supported_types = [tf.float16]  # Use FP16 for smaller size
    
    # Convert
    tflite_model = converter.convert()
    
    # Save
    with open(f'{model_name}.tflite', 'wb') as f:
        f.write(tflite_model)
    
    print(f"✅ Converted {model_name} to TFLite")
    print(f"   Size: {len(tflite_model) / 1024 / 1024:.2f} MB")
